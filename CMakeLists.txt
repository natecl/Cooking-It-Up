cmake_minimum_required(VERSION 3.16)
project(CookingItUp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior checks" OFF)
add_library(cooking DataLoader.cpp Jaccard.cpp Cosine.cpp Ranking.cpp MissingIngredients.cpp)
target_include_directories(cooking PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(cooking PRIVATE /W4)
else()
  target_compile_options(cooking PRIVATE -Wall -Wextra -Wpedantic)
  if(ENABLE_SANITIZERS)
    target_compile_options(cooking PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cooking PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(cooking_it_up main.cpp Menu.cpp Benchmark.cpp)
target_link_libraries(cooking_it_up PRIVATE cooking)
include(CTest)
if(BUILD_TESTING)
  add_executable(cooking_tests tests/tests.cpp)
  target_link_libraries(cooking_tests PRIVATE cooking)
  add_test(NAME cooking_tests COMMAND cooking_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:cooking_it_up> ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
endif()
